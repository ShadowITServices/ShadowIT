# Intune

Endpoint management via **Microsoft Graph** (`DeviceManagement*` endpoints). Connect
with the appropriate scopes (see
[Overview & Connection](/docs/technical-library/script-repository/microsoft-365/)).

## Managed device & compliance report

Exports all Intune-managed devices with OS, ownership, compliance state, and last
check-in — the evidence behind the "compliant devices" Conditional Access control in
[MFA & Entra ID Hardening](/docs/technical-library/mfa-entra-hardening). Read-only;
needs `DeviceManagementManagedDevices.Read.All`.

```powershell
#Requires -Version 5.1
#Requires -Modules Microsoft.Graph.Authentication
<#
.SYNOPSIS
    Reports Intune-managed devices and their compliance state.
.NOTES
    Read-only. Connect-MgGraph -Scopes 'DeviceManagementManagedDevices.Read.All'
    Logs to C:\Temp\IntuneDevices_<timestamp>.log
#>
[CmdletBinding()]
param([int]$StaleCheckinDays = 30)

$ErrorActionPreference = 'Stop'
$stamp   = Get-Date -Format 'yyyyMMdd_HHmmss'
$LogDir  = 'C:\Temp'
$LogFile = Join-Path $LogDir "IntuneDevices_$stamp.log"
$CsvFile = Join-Path $LogDir "IntuneDevices_$stamp.csv"
if (-not (Test-Path -LiteralPath $LogDir)) { New-Item -ItemType Directory -Path $LogDir -Force | Out-Null }
function Write-Log { param($m,$l='INFO')
    $line = '{0} [{1}] {2}' -f (Get-Date -Format 'yyyy-MM-dd HH:mm:ss'), $l, $m
    if ($l -eq 'WARN') { Write-Host $line -ForegroundColor Yellow } else { Write-Host $line }
    Add-Content -LiteralPath $LogFile -Value $line }

try {
    if (-not (Get-MgContext)) { throw 'Not connected. Run Connect-MgGraph with device scopes first.' }
    $cutoff = (Get-Date).AddDays(-$StaleCheckinDays)
    Write-Log 'Retrieving managed devices via Graph.'
    # Paged Graph call for all managed devices.
    $devices = @()
    $uri = 'https://graph.microsoft.com/v1.0/deviceManagement/managedDevices'
    do {
        $resp = Invoke-MgGraphRequest -Method GET -Uri $uri
        $devices += $resp.value
        $uri = $resp.'@odata.nextLink'
    } while ($uri)

    Write-Log "Processing $($devices.Count) device(s)."
    $report = foreach ($d in $devices) {
        $last = [datetime]$d.lastSyncDateTime
        if ($last -lt $cutoff)              { Write-Log "Stale check-in: $($d.deviceName) ($last)" -Level WARN }
        if ($d.complianceState -ne 'compliant') { Write-Log "Non-compliant: $($d.deviceName) ($($d.complianceState))" -Level WARN }
        [pscustomobject]@{
            DeviceName = $d.deviceName; OS = $d.operatingSystem; OSVersion = $d.osVersion
            Owner = $d.managedDeviceOwnerType; Compliance = $d.complianceState
            User = $d.userPrincipalName; LastSync = $last
        }
    }
    $report | Sort-Object Compliance, LastSync | Export-Csv -LiteralPath $CsvFile -NoTypeInformation -Encoding UTF8
    Write-Log "Done. $(($report | Where-Object {$_.Compliance -ne 'compliant'}).Count) non-compliant. CSV: $CsvFile"
}
catch {
    Write-Log "FAILED: $($_.Exception.Message)" -Level ERROR
    throw
}
```

## Export non-compliant devices only

A focused remediation list: just the devices not in a compliant state, with the user to
chase. Read-only; same scope as above.

```powershell
#Requires -Version 5.1
#Requires -Modules Microsoft.Graph.Authentication
<#
.SYNOPSIS
    Exports only non-compliant Intune devices for remediation.
.NOTES
    Read-only. Connect-MgGraph -Scopes 'DeviceManagementManagedDevices.Read.All'
    Logs to C:\Temp\IntuneNonCompliant_<timestamp>.log
#>
[CmdletBinding()]
param()

$ErrorActionPreference = 'Stop'
$stamp   = Get-Date -Format 'yyyyMMdd_HHmmss'
$LogDir  = 'C:\Temp'
$LogFile = Join-Path $LogDir "IntuneNonCompliant_$stamp.log"
$CsvFile = Join-Path $LogDir "IntuneNonCompliant_$stamp.csv"
if (-not (Test-Path -LiteralPath $LogDir)) { New-Item -ItemType Directory -Path $LogDir -Force | Out-Null }
function Write-Log { param($m,$l='INFO')
    $line = '{0} [{1}] {2}' -f (Get-Date -Format 'yyyy-MM-dd HH:mm:ss'), $l, $m
    Write-Host $line; Add-Content -LiteralPath $LogFile -Value $line }

try {
    if (-not (Get-MgContext)) { throw 'Not connected. Run Connect-MgGraph with device scopes first.' }
    Write-Log 'Querying non-compliant devices.'
    $uri = "https://graph.microsoft.com/v1.0/deviceManagement/managedDevices?`$filter=complianceState eq 'noncompliant'"
    $resp = Invoke-MgGraphRequest -Method GET -Uri $uri
    $report = $resp.value | ForEach-Object {
        [pscustomobject]@{ DeviceName=$_.deviceName; OS=$_.operatingSystem; User=$_.userPrincipalName; LastSync=$_.lastSyncDateTime }
    }
    $report | Export-Csv -LiteralPath $CsvFile -NoTypeInformation -Encoding UTF8
    Write-Log "Done. $($report.Count) non-compliant device(s). CSV: $CsvFile"
}
catch {
    Write-Log "FAILED: $($_.Exception.Message)" -Level ERROR
    throw
}
```
